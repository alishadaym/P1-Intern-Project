CREATE TABLE floors (
    id INT AUTO_INCREMENT PRIMARY KEY,
    floor_name VARCHAR(50) NOT NULL,
    floor_code VARCHAR(10) NOT NULL UNIQUE
);

//table for QR locations
CREATE TABLE locations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    location_name VARCHAR(100) NOT NULL,
    location_code VARCHAR(50) NOT NULL UNIQUE,
    floor_id INT NOT NULL,
    x_position INT NOT NULL,
    y_position INT NOT NULL,

    FOREIGN KEY (floor_id) REFERENCES floors(id)
);

//table for stores
CREATE TABLE shops (
    id INT AUTO_INCREMENT PRIMARY KEY,
    shop_name VARCHAR(100) NOT NULL,
    category VARCHAR(100),
    unit VARCHAR(50),
    description TEXT,
    floor_id INT NOT NULL,
    x_position INT NOT NULL,
    y_position INT NOT NULL,

    FOREIGN KEY (floor_id) REFERENCES floors(id)
);

//feedback from mall visitors (complaints/suggestions/compliments about
//facilities) submitted via the guest-facing "Give Feedback" button -
//not tied to a specific shop, so no foreign key to shops
CREATE TABLE feedback (
    id INT AUTO_INCREMENT PRIMARY KEY,
    category VARCHAR(50),
    message TEXT NOT NULL,
    contact VARCHAR(150),
    submitted_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
