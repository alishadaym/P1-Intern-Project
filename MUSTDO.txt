-- Facilities are separate from shops because their availability changes independently.
CREATE TABLE IF NOT EXISTS utilities (
    id INT AUTO_INCREMENT PRIMARY KEY,
    utility_code VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    utility_type ENUM('restroom', 'baby_diaper', 'oku', 'lift') NOT NULL,
    floor_id INT NOT NULL,
    x_position INT NOT NULL,
    y_position INT NOT NULL,
    node_id VARCHAR(50) NOT NULL,
    total_cubicles INT NULL,
    occupied_cubicles INT NULL DEFAULT 0,
    is_occupied BOOLEAN NULL,
    FOREIGN KEY (floor_id) REFERENCES floors(id),
    CHECK (total_cubicles IS NULL OR total_cubicles >= 0),
    CHECK (occupied_cubicles IS NULL OR occupied_cubicles >= 0),
    CHECK (total_cubicles IS NULL OR occupied_cubicles <= total_cubicles)
);
